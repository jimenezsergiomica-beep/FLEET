/**

 */
function crearEstructuraCarpetas() {
  const rootFolder = DriveApp.getRootFolder();
  
  // Carpeta principal
  let carpetaPrincipal = buscarOCrearCarpeta('AVA_FleetControl', rootFolder);
  Logger.log('  📂 Carpeta principal: AVA_FleetControl');
  
  // Subcarpetas organizadas
  buscarOCrearCarpeta('Reportes_BI', carpetaPrincipal);
  buscarOCrearCarpeta('Inspecciones_PDF', carpetaPrincipal);
  buscarOCrearCarpeta('Ordenes_Trabajo_PDF', carpetaPrincipal);
  buscarOCrearCarpeta('Documentos_Unidades', carpetaPrincipal);
  
  Logger.log('  ├─ Reportes_BI/');
  Logger.log('  ├─ Inspecciones_PDF/');
  Logger.log('  ├─ Ordenes_Trabajo_PDF/');
  Logger.log('  └─ Documentos_Unidades/');
}

function buscarOCrearCarpeta(nombre, parentFolder) {
  const folders = parentFolder.getFoldersByName(nombre);
  if(folders.hasNext()) {
    return folders.next();
  } else {
    return parentFolder.createFolder(nombre);
  }
}

 * ðŸ”§ CONFIGURAR HOJA UNIDADES
 * Ejecutar manualmente desde el Editor de Apps Script.
 * Crea las columnas necesarias en orden correcto para el expediente de unidades.
 */
function CONFIGURAR_HOJA_UNIDADES() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sheet = ss.getSheetByName('UNIDADES');
  
  // Si no existe, crearla
  if(!sheet) {
    sheet = ss.insertSheet('UNIDADES');
    Logger.log('âœ… Hoja UNIDADES creada');
  }
  
  // Columnas requeridas en orden
  const columnasRequeridas = [
    'ID',              // A - Identificador Ãºnico de unidad
    'MODELO',          // B - Modelo del vehÃ­culo
    'PLACAS',          // C - Placas
    'ESTADO',          // D - DISPONIBLE, EN_VIAJE, MANTENIMIENTO
    'ODOMETRO',        // E - KilÃ³metros actuales
    'MARCA',           // F - Marca del vehÃ­culo
    'AÃ‘O',             // G - AÃ±o del vehÃ­culo
    'COLOR',           // H - Color
    'NO_SERIE',        // I - NÃºmero de serie/VIN
    'NO_MOTOR',        // J - NÃºmero de motor
    'TARJETA_VIG',     // K - Vigencia tarjeta de circulaciÃ³n
    'SEGURO_VIG',      // L - Vigencia del seguro
    'ASEGURADORA',     // M - Nombre de aseguradora
    'POLIZA',          // N - NÃºmero de pÃ³liza
    'VERIFICACION_VIG',// O - Vigencia verificaciÃ³n vehicular
    'FOTOS',           // P - JSON con base64 de fotos (mÃ¡x 3)
    'PERMISOS',        // Q - JSON con permisos especiales
    'DOCUMENTOS',      // R - JSON con documentos escaneados
    'ALIAS',           // S - Alias/apodo opcional de la unidad
    'OPERADOR_ASIGNADO'// T - Operador/chofer asignado a la unidad
  ];
  
  // Obtener encabezados actuales
  const data = sheet.getDataRange().getValues();
  const headersActuales = data[0] || [];
  
  // Verificar si ya estÃ¡n configuradas correctamente
  let necesitaReconfigurar = false;
  if(headersActuales.length < columnasRequeridas.length) {
    necesitaReconfigurar = true;
  } else {
    for(let i=0; i<columnasRequeridas.length; i++) {
      if(headersActuales[i] !== columnasRequeridas[i]) {
        necesitaReconfigurar = true;
        break;
      }
    }
  }
  
  if(!necesitaReconfigurar) {
    Logger.log('âœ… Hoja UNIDADES ya estÃ¡ configurada correctamente');
    return;
  }
  
  // Si hay datos, necesitamos reorganizar
  if(data.length > 1) {
    Logger.log('âš ï¸ La hoja tiene datos. Agregando columnas faltantes...');
    
    // Crear mapa de columnas actuales
    const mapActual = {};
    headersActuales.forEach((h, i) => mapActual[h] = i);
    
    // Agregar columnas faltantes al final
    let colNueva = headersActuales.length + 1;
    columnasRequeridas.forEach(col => {
      if(mapActual[col] === undefined) {
        sheet.getRange(1, colNueva).setValue(col);
        Logger.log(`âž• Columna ${col} agregada en posiciÃ³n ${colNueva}`);
        colNueva++;
      }
    });
    
    Logger.log('âœ… Columnas faltantes agregadas. Reordene manualmente si lo desea.');
  } else {
    // Hoja vacÃ­a - configurar desde cero
    sheet.getRange(1, 1, 1, columnasRequeridas.length).setValues([columnasRequeridas]);
    
    // Formato de encabezados
    const headerRange = sheet.getRange(1, 1, 1, columnasRequeridas.length);
    headerRange.setBackground('#1a1a2e')
               .setFontColor('#ffffff')
               .setFontWeight('bold')
               .setHorizontalAlignment('center');
    
    // Congelar primera fila
    sheet.setFrozenRows(1);
    
    // Ajustar anchos
    sheet.setColumnWidth(1, 100);  // ID
    sheet.setColumnWidth(2, 120);  // MODELO
    sheet.setColumnWidth(15, 50);  // FOTOS (contendrÃ¡ JSON largo)
    sheet.setColumnWidth(16, 50);  // PERMISOS (contendrÃ¡ JSON largo)
    
    Logger.log('âœ… Hoja UNIDADES configurada con ' + columnasRequeridas.length + ' columnas');
  }
  
  Logger.log('ðŸ“‹ Columnas: ' + columnasRequeridas.join(', '));

function validarPin(pin) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sheet = ss.getSheetByName('USUARIOS');
  if (!sheet) return { exito: false, error: 'Base de datos no conectada' };
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][0]).trim() == String(pin).trim() && (data[i][3] === 'ACTIVO' || !data[i][3])) {
      return { exito: true, nombre: data[i][1], rol: data[i][2], pin: data[i][0] };
    }
  }
  return { exito: false };
}

function obtenerDatosFlota() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sU = ss.getSheetByName('UNIDADES');
  const sV = ss.getSheetByName('VIAJES');
  if(!sU) return [];
  
  const uData = sU.getDataRange().getValues();
  const vData = sV ? sV.getDataRange().getValues() : [];
  const headers = uData[0];
  
  // FunciÃ³n para obtener Ã­ndice de columna
  const getCol = (name) => headers.indexOf(name);
  
  let unidades = [];
  for(let i=1; i<uData.length; i++){
    // Obtener foto principal (primera del array de fotos)
    let fotoBase64 = '';
    try {
      const fotosCol = getCol('FOTOS');
      if(fotosCol >= 0 && uData[i][fotosCol]) {
        const fotosArr = JSON.parse(uData[i][fotosCol]);
        if(fotosArr && fotosArr.length > 0) {
          fotoBase64 = fotosArr[0]; // Primera foto
        }
      }
    } catch(e) { fotoBase64 = ''; }
    
    // Obtener alias
    const aliasCol = getCol('ALIAS');
    const alias = aliasCol >= 0 ? (uData[i][aliasCol] || '') : '';
    
    // Obtener operador asignado (de la columna o del viaje activo)
    const operadorCol = getCol('OPERADOR_ASIGNADO');
    let operadorAsignado = operadorCol >= 0 ? (uData[i][operadorCol] || '') : '';
    
    let u = { 
      id: uData[i][0], 
      placas: uData[i][getCol('PLACAS')] || uData[i][2] || '', 
      modelo: uData[i][getCol('MODELO')] || uData[i][1] || '', 
      estado: uData[i][getCol('ESTADO')] || uData[i][3] || 'DISPONIBLE', 
      chofer: operadorAsignado || 'Sin Asignar', 
      foto: fotoBase64,
      alias: alias
    };
    
    // Si estÃ¡ en viaje, obtener el chofer del viaje activo
    if(u.estado === 'EN_VIAJE'){
      for(let j=vData.length-1; j>0; j--){
        if(vData[j][1] === u.id && vData[j][11] === 'ABIERTO'){
          u.chofer = obtenerNombre(vData[j][2]); break;
        }
      }
    }
    unidades.push(u);
  }
  return unidades;
}

function obtenerDatosFlotaCompleto() { return obtenerDatosFlota(); } // Alias para seguridad

function obtenerExpedienteUnidad(unidadId) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sU = ss.getSheetByName('UNIDADES');
  if(!sU) return null;
  
  const data = sU.getDataRange().getValues();
  const headers = data[0];
  
  for(let i=1; i<data.length; i++) {
    if(data[i][0] === unidadId) {
      // Buscar Ã­ndices de columnas (flexibilidad)
      const getCol = (name) => headers.indexOf(name);
      const getVal = (name, fallback) => {
        const idx = getCol(name);
        return idx >= 0 ? (data[i][idx] || fallback) : fallback;
      };
      
      // Parsear JSON de forma segura
      const parseJSON = (name, defaultVal) => {
        try {
          const idx = getCol(name);
          if(idx >= 0 && data[i][idx]) {
            return JSON.parse(data[i][idx]);
          }
          return defaultVal;
        } catch(e) { return defaultVal; }
      };
      
      return {
        id: data[i][0],
        modelo: data[i][1] || '',
        placas: data[i][2] || '',
        estado: data[i][3] || '',
        odometro: data[i][4] || 0,
        marca: getVal('MARCA', data[i][5] || ''),
        anio: getVal('AÃ‘O', data[i][6] || ''),
        color: getVal('COLOR', data[i][7] || ''),
        noSerie: getVal('NO_SERIE', data[i][8] || ''),
        noMotor: getVal('NO_MOTOR', data[i][9] || ''),
        tarjetaVig: getVal('TARJETA_VIG', ''),
        seguroVig: getVal('SEGURO_VIG', ''),
        aseguradora: getVal('ASEGURADORA', ''),
        poliza: getVal('POLIZA', ''),
        verificacionVig: getVal('VERIFICACION_VIG', ''),
        fotos: parseJSON('FOTOS', []),
        permisos: parseJSON('PERMISOS', []),
        documentos: parseJSON('DOCUMENTOS', {}),
        alias: getVal('ALIAS', ''),
        operadorAsignado: getVal('OPERADOR_ASIGNADO', '')
      };
    }
  }
  return null;
}

function guardarExpedienteUnidad(expediente) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let sU = ss.getSheetByName('UNIDADES');
  if(!sU) return { exito: false, error: 'Hoja UNIDADES no encontrada' };
  
  const data = sU.getDataRange().getValues();
  let headers = data[0] || [];
  
  // Columnas requeridas (incluyendo nuevas)
  const columnas = ['ID','MODELO','PLACAS','ESTADO','ODOMETRO','MARCA','AÃ‘O','COLOR','NO_SERIE','NO_MOTOR','TARJETA_VIG','SEGURO_VIG','ASEGURADORA','POLIZA','VERIFICACION_VIG','FOTOS','PERMISOS','DOCUMENTOS','ALIAS','OPERADOR_ASIGNADO'];
  
  // Asegurar columnas existen
  columnas.forEach((col) => {
    if(headers.indexOf(col) === -1) {
      sU.getRange(1, headers.length + 1).setValue(col);
      headers.push(col);
    }
  });
  
  // Recargar headers despuÃ©s de agregar columnas
  headers = sU.getRange(1, 1, 1, sU.getLastColumn()).getValues()[0];
  
  const getCol = (name) => headers.indexOf(name) + 1;
  
  // Buscar si existe la unidad
  let filaExistente = -1;
  for(let i=1; i<data.length; i++) {
    if(data[i][0] === expediente.id) {
      filaExistente = i + 1;
      break;
    }
  }
  
  if(filaExistente > 0) {
    // ACTUALIZAR unidad existente
    const row = filaExistente;
    
    if(getCol('PLACAS') > 0) sU.getRange(row, getCol('PLACAS')).setValue(expediente.placas);
    if(getCol('MARCA') > 0) sU.getRange(row, getCol('MARCA')).setValue(expediente.marca);
    if(getCol('MODELO') > 0) sU.getRange(row, getCol('MODELO')).setValue(expediente.modelo);
    if(getCol('AÃ‘O') > 0) sU.getRange(row, getCol('AÃ‘O')).setValue(expediente.anio);
    if(getCol('COLOR') > 0) sU.getRange(row, getCol('COLOR')).setValue(expediente.color);
    if(getCol('ODOMETRO') > 0) sU.getRange(row, getCol('ODOMETRO')).setValue(expediente.odometro);
    if(getCol('NO_SERIE') > 0) sU.getRange(row, getCol('NO_SERIE')).setValue(expediente.noSerie);
    if(getCol('NO_MOTOR') > 0) sU.getRange(row, getCol('NO_MOTOR')).setValue(expediente.noMotor);
    if(getCol('TARJETA_VIG') > 0) sU.getRange(row, getCol('TARJETA_VIG')).setValue(expediente.tarjetaVig);
    if(getCol('SEGURO_VIG') > 0) sU.getRange(row, getCol('SEGURO_VIG')).setValue(expediente.seguroVig);
    if(getCol('ASEGURADORA') > 0) sU.getRange(row, getCol('ASEGURADORA')).setValue(expediente.aseguradora);
    if(getCol('POLIZA') > 0) sU.getRange(row, getCol('POLIZA')).setValue(expediente.poliza);
    if(getCol('VERIFICACION_VIG') > 0) sU.getRange(row, getCol('VERIFICACION_VIG')).setValue(expediente.verificacionVig || '');
    if(getCol('FOTOS') > 0) sU.getRange(row, getCol('FOTOS')).setValue(JSON.stringify(expediente.fotos || []));
    if(getCol('PERMISOS') > 0) sU.getRange(row, getCol('PERMISOS')).setValue(JSON.stringify(expediente.permisos || []));
    if(getCol('DOCUMENTOS') > 0) sU.getRange(row, getCol('DOCUMENTOS')).setValue(JSON.stringify(expediente.documentos || {}));
    if(getCol('ALIAS') > 0) sU.getRange(row, getCol('ALIAS')).setValue(expediente.alias || '');
    if(getCol('OPERADOR_ASIGNADO') > 0) sU.getRange(row, getCol('OPERADOR_ASIGNADO')).setValue(expediente.operadorAsignado || '');
    
    return { exito: true, mensaje: 'Unidad actualizada' };
  } else {
    // CREAR nueva unidad
    if(!expediente.id || expediente.id.trim() === '') {
      return { exito: false, error: 'ID de unidad requerido' };
    }
    
    // Verificar que no exista ya
    for(let i=1; i<data.length; i++) {
      if(data[i][0] === expediente.id) {
        return { exito: false, error: 'Ya existe una unidad con ese ID' };
      }
    }
    
    // Crear fila nueva
    const nuevaFila = [];
    columnas.forEach(col => {
      switch(col) {
        case 'ID': nuevaFila.push(expediente.id); break;
        case 'MODELO': nuevaFila.push(expediente.modelo || ''); break;
        case 'PLACAS': nuevaFila.push(expediente.placas || ''); break;
        case 'ESTADO': nuevaFila.push('DISPONIBLE'); break;
        case 'ODOMETRO': nuevaFila.push(expediente.odometro || 0); break;
        case 'MARCA': nuevaFila.push(expediente.marca || ''); break;
        case 'AÃ‘O': nuevaFila.push(expediente.anio || ''); break;
        case 'COLOR': nuevaFila.push(expediente.color || ''); break;
        case 'NO_SERIE': nuevaFila.push(expediente.noSerie || ''); break;
        case 'NO_MOTOR': nuevaFila.push(expediente.noMotor || ''); break;
        case 'TARJETA_VIG': nuevaFila.push(expediente.tarjetaVig || ''); break;
        case 'SEGURO_VIG': nuevaFila.push(expediente.seguroVig || ''); break;
        case 'ASEGURADORA': nuevaFila.push(expediente.aseguradora || ''); break;
        case 'POLIZA': nuevaFila.push(expediente.poliza || ''); break;
        case 'VERIFICACION_VIG': nuevaFila.push(expediente.verificacionVig || ''); break;
        case 'FOTOS': nuevaFila.push(JSON.stringify(expediente.fotos || [])); break;
        case 'PERMISOS': nuevaFila.push(JSON.stringify(expediente.permisos || [])); break;
        case 'DOCUMENTOS': nuevaFila.push(JSON.stringify(expediente.documentos || {})); break;
        case 'ALIAS': nuevaFila.push(expediente.alias || ''); break;
        case 'OPERADOR_ASIGNADO': nuevaFila.push(expediente.operadorAsignado || ''); break;
        default: nuevaFila.push('');
      }
    });
    
    sU.appendRow(nuevaFila);
    return { exito: true, mensaje: 'Nueva unidad creada', id: expediente.id };
  }
}

function obtenerHistorialOTsUnidad(unidadId) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('ORDENES_TRABAJO');
  if(!s) return [];
  
  const data = s.getDataRange().getValues();
  let lista = [];
  
  for(let i=data.length-1; i>0 && lista.length<50; i--) {
    if(data[i][2] === unidadId) {
      lista.push({
        folio: data[i][0],
        fecha: data[i][1] ? new Date(data[i][1]).toLocaleDateString() : '',
        tipo: data[i][4],
        actividad: data[i][3],
        diagnostico: data[i][10] || '',
        estado: data[i][5],
        pdfUrl: data[i][13] || ''
      });
    }
  }
  
  return lista;
}

function obtenerCostosUnidad(unidadId) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('ORDENES_TRABAJO');
  if(!s) return { totalCosto: 0, totalOTs: 0, totalHoras: 0, lista: [] };
  
  const data = s.getDataRange().getValues();
  let totalCosto = 0, totalOTs = 0, totalHoras = 0, lista = [];
  
  for(let i=data.length-1; i>0; i--) {
    if(data[i][2] === unidadId && data[i][5] === 'CERRADA') {
      const costo = Number(data[i][11]) || 0;
      const horas = Number(data[i][12]) || 0;
      
      totalCosto += costo;
      totalHoras += horas;
      totalOTs++;
      
      if(lista.length < 30) {
        lista.push({
          folio: data[i][0],
          fecha: data[i][1] ? new Date(data[i][1]).toLocaleDateString() : '',
          tipo: data[i][4],
          actividad: data[i][3],
          horas: horas,
          costo: costo
        });
      }
    }
  }
  
  return { totalCosto, totalOTs, totalHoras, lista };
}

function obtenerPreventivosUnidad(unidadId) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('ORDENES_TRABAJO');
  if(!s) return { proximo: null, historial: [] };
  
  const data = s.getDataRange().getValues();
  let proximo = null, historial = [];
  
  for(let i=1; i<data.length; i++) {
    if(data[i][2] === unidadId && data[i][4] === 'PREVENTIVO') {
      const ot = {
        folio: data[i][0],
        fecha: data[i][7] ? new Date(data[i][7]).toLocaleDateString() : new Date(data[i][1]).toLocaleDateString(),
        actividad: data[i][3],
        estado: data[i][5],
        tecnico: data[i][6]
      };
      
      if(data[i][5] === 'PROGRAMADA' || data[i][5] === 'ESPERA_MAT') {
        if(!proximo) proximo = ot;
      } else if(data[i][5] === 'CERRADA') {
        if(historial.length < 20) historial.push(ot);
      }
    }
  }
  
  // Ordenar historial por fecha descendente
  historial.sort((a, b) => new Date(b.fecha) - new Date(a.fecha));
  
  return { proximo, historial };
}

// ============== CAPTURA DE HISTORIAL (EXPEDIENTES EN PAPEL) ==============
function guardarOTHistorica(ot) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let s = ss.getSheetByName('ORDENES_TRABAJO');
  
  if(!s) {
    s = ss.insertSheet('ORDENES_TRABAJO');
    s.getRange(1, 1, 1, 14).setValues([['FOLIO', 'FECHA', 'UNIDAD', 'ACTIVIDAD', 'TIPO', 'ESTADO', 'TECNICO', 'FECHA_PROG', 'INICIO_REAL', 'FIN_REAL', 'DIAGNOSTICO', 'COSTO', 'HORAS', 'PDF_URL']]);
  }
  
  // Verificar si el folio ya existe
  const data = s.getDataRange().getValues();
  for(let i=1; i<data.length; i++) {
    if(data[i][0] === ot.folio) {
      // Actualizar OT existente (ediciÃ³n)
      const row = i + 1;
      s.getRange(row, 2).setValue(ot.fecha);
      s.getRange(row, 4).setValue(ot.actividad);
      s.getRange(row, 5).setValue(ot.tipo);
      s.getRange(row, 7).setValue(ot.tecnico);
      s.getRange(row, 11).setValue(ot.diagnostico);
      s.getRange(row, 12).setValue(ot.costo);
      s.getRange(row, 13).setValue(ot.horas);
      return { exito: true, mensaje: 'OT actualizada', folio: ot.folio };
    }
  }
  
  // OT Nueva - crear fila
  const nuevaFila = [
    ot.folio,                    // FOLIO
    ot.fecha,                    // FECHA
    ot.unidad,                   // UNIDAD
    ot.actividad,                // ACTIVIDAD
    ot.tipo,                     // TIPO
    'CERRADA',                   // ESTADO (histÃ³rica = cerrada)
    ot.tecnico || 'HISTÃ“RICO',   // TECNICO
    ot.fecha,                    // FECHA_PROG
    ot.fecha,                    // INICIO_REAL
    ot.fecha,                    // FIN_REAL
    ot.diagnostico,              // DIAGNOSTICO
    ot.costo || 0,               // COSTO
    ot.horas || 0,               // HORAS
    ''                           // PDF_URL
  ];
  
  s.appendRow(nuevaFila);
  return { exito: true, mensaje: 'OT histÃ³rica registrada', folio: ot.folio };
}

// --- 4. TALLER: PROGRAMACIÃ“N Y OT ---
function guardarProgramacion(datos) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  
  // 1. Crear OT
  let sOT = ss.getSheetByName('ORDENES_TRABAJO');
  if(!sOT) { 
    sOT=ss.insertSheet('ORDENES_TRABAJO'); 
    sOT.appendRow(['FOLIO','FECHA','UNIDAD','ACTIVIDAD','TIPO','ESTADO','TECNICO','FECHA_PROG','INICIO_REAL','FIN_REAL','DIAGNOSTICO','COSTO','HORAS','PDF_URL']); 
  }
  
  // Generar folio con formato legible: OT-TIPO-YYYYMMDD-XXX
  const hoy = new Date();
  const fechaStr = hoy.toISOString().split('T')[0].replace(/-/g,'');
  const folioOT = `OT-${datos.tipo.substring(0,4)}-${fechaStr}-${Date.now().toString().slice(-4)}`;
  const estadoInicial = (datos.insumos || datos.refacciones) ? 'ESPERA_MAT' : 'PROGRAMADA';
  
  sOT.appendRow([folioOT, hoy, datos.unidad, datos.actividad, datos.tipo, estadoInicial, datos.tecnico, datos.fechaProg, '', '', '', 0, 0, '']);
  
  // 2. Crear Solicitud (si aplica)
  let folioSol = '';
  if (datos.insumos || datos.refacciones) {
    let sSol = ss.getSheetByName('SOLICITUDES');
    if(!sSol) { sSol=ss.insertSheet('SOLICITUDES'); sSol.appendRow(['FOLIO','OT_REF','UNIDAD','INSUMOS','REFACCIONES','ESTADO','COSTO','FECHA_ENTREGA']); }
    
    folioSol = 'SOL-' + Date.now();
    sSol.appendRow([folioSol, folioOT, datos.unidad, datos.insumos, datos.refacciones, 'PENDIENTE', 0, '']);
    
    registrarTrazabilidad(datos.unidad, folioOT, 'SOLICITUD_CREADA', `Solicitud ${folioSol} generada por Taller.`, datos.tecnico);
  }

  // 3. Trazabilidad General
  registrarTrazabilidad(datos.unidad, folioOT, 'PROGRAMACION', `Mantenimiento ${datos.tipo} agendado para ${datos.fechaProg}.`, datos.tecnico);
  registrarEventoKardex(datos.unidad, folioOT, datos.tipo, datos.actividad);

  // 4. Bloquear unidad si es para hoy
  const prog = new Date(datos.fechaProg);
  if(prog <= hoy) actualizarEstadoUnidad(datos.unidad, 'MANTENIMIENTO');

  return { exito: true, folio: folioOT };
}

function obtenerKPIsTaller() {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('ORDENES_TRABAJO');
  if(!s) return { programadas: 0, esperaMat: 0, enProceso: 0, cerradasHoy: 0 };
  
  const data = s.getDataRange().getValues();
  let programadas = 0, esperaMat = 0, enProceso = 0, cerradasHoy = 0;
  const hoy = new Date().toDateString();
  
  for(let i=1; i<data.length; i++) {
    const estado = data[i][5];
    if(estado === 'PROGRAMADA') programadas++;
    else if(estado === 'ESPERA_MAT') esperaMat++;
    else if(estado === 'EN_PROCESO') enProceso++;
    else if(estado === 'CERRADA') {
      const fechaCierre = data[i][9];
      if(fechaCierre && new Date(fechaCierre).toDateString() === hoy) cerradasHoy++;
    }
  }
  
  return { programadas, esperaMat, enProceso, cerradasHoy };
}

function actualizarEstadoOT(folio, nuevoEstado) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('ORDENES_TRABAJO');
  if(!s) return { exito: false, error: 'Hoja no encontrada' };
  
  const data = s.getDataRange().getValues();
  for(let i=1; i<data.length; i++) {
    if(data[i][0] === folio) {
      s.getRange(i+1, 6).setValue(nuevoEstado); // Estado columna 6
      
      // Si inicia trabajo, registrar hora inicio
      if(nuevoEstado === 'EN_PROCESO' && !data[i][8]) {
        s.getRange(i+1, 9).setValue(new Date()); // Inicio real columna 9
      }
      
      // Registrar trazabilidad
      registrarTrazabilidad(data[i][2], folio, 'CAMBIO_ESTADO', `Estado cambiado a ${nuevoEstado}`, 'TALLER');
      
      return { exito: true };
    }
  }
  return { exito: false, error: 'OT no encontrada' };
}

function cerrarOrdenTrabajo(folio, diagnostico, costo, horas) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('ORDENES_TRABAJO');
  if(!s) return { exito: false, error: 'Hoja no encontrada' };
  
  const data = s.getDataRange().getValues();
  for(let i=1; i<data.length; i++) {
    if(data[i][0] === folio) {
      const unidad = data[i][2];
      const ahora = new Date();
      
      // Actualizar registro
      s.getRange(i+1, 6).setValue('CERRADA');           // Estado
      s.getRange(i+1, 10).setValue(ahora);              // Fin real
      s.getRange(i+1, 11).setValue(diagnostico);        // DiagnÃ³stico
      s.getRange(i+1, 12).setValue(Number(costo) || 0); // Costo
      s.getRange(i+1, 13).setValue(Number(horas) || 0); // Horas
      
      // Generar PDF
      const pdfUrl = generarPDFOrdenTrabajoInterno(folio, {
        folio: folio,
        unidad: unidad,
        actividad: data[i][3],
        tipo: data[i][4],
        tecnico: data[i][6],
        fechaProg: data[i][7],
        inicioReal: data[i][8],
        finReal: ahora,
        diagnostico: diagnostico,
        costo: costo,
        horas: horas
      });
      
      // Guardar URL del PDF
      s.getRange(i+1, 14).setValue(pdfUrl);
      
      // Liberar unidad
      actualizarEstadoUnidad(unidad, 'DISPONIBLE');
      
      // Trazabilidad
      registrarTrazabilidad(unidad, folio, 'OT_CERRADA', `OT cerrada. DiagnÃ³stico: ${diagnostico.substring(0,50)}...`, 'TALLER');
      
      return { exito: true, pdfUrl: pdfUrl };
    }
  }
  return { exito: false, error: 'OT no encontrada' };
}

function gestionarSolicitud(folioSol, accion, costo) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('SOLICITUDES');
  const d = s.getDataRange().getValues();
  
  for(let i=1; i<d.length; i++){
    if(d[i][0] == folioSol){
      const otRef = d[i][1];
      const unidad = d[i][2];
      
      if(accion === 'APROBAR') {
        s.getRange(i+1, 6).setValue('APROBADA');
        registrarTrazabilidad(unidad, otRef, 'SOLICITUD_APROBADA', 'Material autorizado para compra.', 'ADMIN');
      } 
      else if(accion === 'ENTREGAR') {
        s.getRange(i+1, 6).setValue('ENTREGADA');
        s.getRange(i+1, 7).setValue(costo);
        s.getRange(i+1, 8).setValue(new Date());
        
        // Reactivar OT
        const sOT = ss.getSheetByName('ORDENES_TRABAJO');
        const dOT = sOT.getDataRange().getValues();
        for(let j=1; j<dOT.length; j++){
          if(dOT[j][0] == otRef){
            sOT.getRange(j+1, 6).setValue('EN_PROCESO'); // Estado
            sOT.getRange(j+1, 9).setValue(new Date());   // Inicio Real
            break;
          }
        }
        registrarTrazabilidad(unidad, otRef, 'ENTREGA_MATERIAL', `Material entregado ($${costo}). Inicia trabajo en taller.`, 'ADMIN');
      }
      break;
    }
  }
  return { exito: true };
}

function procesarEntrada(datos) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const sv = ss.getSheetByName('VIAJES');
  const data = sv.getDataRange().getValues();
  let fila = -1;
  for(let i=data.length-1; i>0; i--){
    if(data[i][1] == datos.unidad && data[i][11] == 'ABIERTO') { fila = i+1; break; }
  }
  
  if(fila > 0){
    sv.getRange(fila, 9).setValue(new Date());
    sv.getRange(fila, 10).setValue(datos.km);
    sv.getRange(fila, 12).setValue('CERRADO');
    if(datos.falla === 'SI'){
      sv.getRange(fila, 13).setValue('SI');
      sv.getRange(fila, 14).setValue(datos.descFalla);
      actualizarEstadoUnidad(datos.unidad, 'MANTENIMIENTO');
      registrarEventoKardex(datos.unidad, 'VIAJE-'+Date.now(), 'CORRECTIVO', 'Falla en ruta: ' + datos.descFalla);
    } else {
      actualizarEstadoUnidad(datos.unidad, 'DISPONIBLE');
    }
    const sUni = ss.getSheetByName('UNIDADES');
    const dUni = sUni.getDataRange().getValues();
    for(let k=1; k<dUni.length; k++) if(dUni[k][0] == datos.unidad) { sUni.getRange(k+1, 5).setValue(datos.km); break; }
    return {exito:true};
  }
  return {exito:false, error: 'No hay viaje abierto'};
}

function getReporteBiOperativo(ini, fin, choferPin) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const vData = ss.getSheetByName('VIAJES').getDataRange().getValues();
  const dIni = new Date(ini), dFin = new Date(fin); 
  dFin.setHours(23,59,59);
  
  let result = [];
  for(let i=1; i<vData.length; i++) {
    const fecha = new Date(vData[i][3]);
    if(fecha >= dIni && fecha <= dFin) {
      // Filtro por chofer si aplica
      if(choferPin && choferPin !== 'TODOS' && String(vData[i][2]) !== String(choferPin)) continue;
      
      const kmSalida = Number(vData[i][4]) || 0;
      const kmEntrada = Number(vData[i][9]) || 0;
      const kmRecorrido = kmEntrada > kmSalida ? kmEntrada - kmSalida : 0;
      
      result.push({
        fecha: fecha.toLocaleDateString(),
        unidad: vData[i][1],
        chofer: obtenerNombre(vData[i][2]),
        motivo: vData[i][5] || 'Sin destino',
        kmRecorrido: kmRecorrido,
        estado: vData[i][11]
      });
    }
  }
  return result;
}

function getReporteBiCombustible(ini, fin, unidadFiltro) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const dSheet = ss.getSheetByName('DIESEL');
  if(!dSheet) return [];
  
  const dData = dSheet.getDataRange().getValues();
  const dIni = new Date(ini), dFin = new Date(fin); 
  dFin.setHours(23,59,59);
  
  // Obtener odÃ³metros para calcular rendimiento
  const uSheet = ss.getSheetByName('UNIDADES');
  let odometros = {};
  if(uSheet) {
    const uData = uSheet.getDataRange().getValues();
    for(let i=1; i<uData.length; i++) {
      odometros[uData[i][0]] = Number(uData[i][4]) || 0;
    }
  }
  
  let result = [];
  let prevKm = {};
  
  for(let i=1; i<dData.length; i++) {
    const fecha = new Date(dData[i][0]);
    if(fecha >= dIni && fecha <= dFin) {
      const unidad = dData[i][1];
      
      // Filtro por unidad si aplica
      if(unidadFiltro && unidadFiltro !== 'TODAS' && unidad !== unidadFiltro) continue;
      
      const litros = Number(dData[i][3]) || 0;
      const costo = Number(dData[i][4]) || 0;
      const kmActual = Number(dData[i][5]) || 0;
      
      // Calcular rendimiento (km/litro)
      let rend = 'N/A';
      if(prevKm[unidad] && kmActual > prevKm[unidad] && litros > 0) {
        rend = ((kmActual - prevKm[unidad]) / litros).toFixed(2);
      }
      prevKm[unidad] = kmActual;
      
      result.push({
        fecha: fecha.toLocaleDateString(),
        unidad: unidad,
        litros: litros,
        costo: costo,
        km: kmActual,
        rend: rend
      });
    }
  }
  return result;
}

function obtenerKPIsTaller() {
    const agenda = obtenerAgendaTaller();
    let prog = 0, espera = 0, proc = 0, cerradas = 0;
    agenda.forEach(a => {
        if(a.estado === 'PROGRAMADA') prog++;
        if(a.estado === 'ESPERA_MAT') espera++;
        if(a.estado === 'EN_PROCESO') proc++;
    });
    
    // Para cerradas HOY
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const s = ss.getSheetByName('ORDENES_TRABAJO');
    if(s) {
        const data = s.getDataRange().getValues();
        const hoy = new Date();
        for(let i=1; i<data.length; i++) {
            if(data[i][5] === 'CERRADA') {
                const fCierre = new Date(data[i][7]); // Usando fecha PROG/CIERRE
                if(fCierre.getDate() == hoy.getDate() && fCierre.getMonth() == hoy.getMonth() && fCierre.getFullYear() == hoy.getFullYear()) {
                    cerradas++;
                }
            }
        }
    }
    return { programadas: prog, esperaMat: espera, enProceso: proc, cerradasHoy: cerradas };
}

function cambiarEstadoSolicitud(id, nuevoEstado, motivo) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('SOLICITUDES');
  if(!s) return { exito: false, error: 'Hoja SOLICITUDES no encontrada' };
  
  let otRef = '';
  
  // 1. Actualizar Solicitud
  const data = s.getDataRange().getValues();
  let encontrada = false;
  for(let i=1; i<data.length; i++) {
    if(data[i][0] === id) {
      s.getRange(i+1, 8).setValue(nuevoEstado); // ESTADO
      if(motivo) s.getRange(i+1, 9).setValue(motivo); // MOTIVO_RECHAZO
      otRef = data[i][1]; // El Folio es la referencia a la OT
      encontrada = true;
      break;
    }
  }
  
  if(!encontrada) return { exito: false, error: 'Solicitud no encontrada' };
  
  // 2. Sincronizar con OT (Si existe)
  if(otRef) {
      const sOT = ss.getSheetByName('ORDENES_TRABAJO');
      if(sOT) {
          const dOT = sOT.getDataRange().getValues();
          for(let i=1; i<dOT.length; i++) {
              if(dOT[i][0] === otRef) {
                  // Si Admin AUTORIZA, la OT pasa a PROGRAMADA (lista para trabajar)
                  // Si Admin RECHAZA, la OT pasa a CANCELADA/RECHAZADA
                  let estadoOT = 'PENDIENTE';
                  if(nuevoEstado === 'AUTORIZADA') estadoOT = 'PROGRAMADA';
                  if(nuevoEstado === 'RECHAZADA') estadoOT = 'RECHAZADA';
                  
                  sOT.getRange(i+1, 6).setValue(estadoOT);
                  break;
              }
          }
      }
  }
  
  return { exito: true };
}

      const ahora = new Date();
      const diff = ahora - inicio;
      const horas = Math.floor(diff / 3600000);
      const minutos = Math.floor((diff % 3600000) / 60000);
      tiempoEntrega = `${horas}h ${minutos}m`;
    }
    
    solicitudes.push({
      id: data[i][0],
      folio: data[i][1],
      unidad: data[i][2],
      tipo: data[i][3],
      descripcion: data[i][4],
      solicitante: data[i][5],
      fecha: data[i][6] ? new Date(data[i][6]).toLocaleDateString() : '',
      estado: estado,
      tiempoEntrega: tiempoEntrega || data[i][11]
    });
  }
  
  // Ordenar por fecha descendente
  solicitudes.sort((a, b) => new Date(b.fecha) - new Date(a.fecha));
  
  return { solicitudes, pendientes, autorizadas, entregadas, rechazadas };
}

/**
 * MODAL 3: REGISTRO DE ENTRADA
 * Cierra viaje, actualiza odómetro y registra fallas
 */
function procesarEntrada(datos) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  const s = ss.getSheetByName('VIAJES');
  
  if(!s) return { exito: false, error: 'Hoja VIAJES no encontrada' };
  
  const data = s.getDataRange().getValues();
  const hora = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'HH:mm:ss');
  
  // Buscar último viaje ABIERTO de esta unidad
  for(let i = data.length - 1; i > 0; i--) {
    if(data[i][2] === datos.unidad && data[i][11] === 'ABIERTO') {
      const row = i + 1;
      
      // Actualizar datos de entrada
      s.getRange(row, 8).setValue(datos.km);  // KM_LLEGADA
      s.getRange(row, 9).setValue(hora);      // HORA_LLEGADA
      s.getRange(row, 10).setValue(datos.falla || 'NO');  // FALLA
      s.getRange(row, 11).setValue(datos.descFalla || '');  // DESC_FALLA
      s.getRange(row, 12).setValue('CERRADO');  // ESTADO
      
      // Actualizar odómetro de la unidad
      const sU = ss.getSheetByName('UNIDADES');
      if(sU) {
        const uData = sU.getDataRange().getValues();
        for(let j = 1; j < uData.length; j++) {
          if(uData[j][0] === datos.unidad) {
            sU.getRange(j + 1, 5).setValue(datos.km);  // ODOMETRO
            break;
          }
        }
      }
      
      // Si reportó falla, cambiar estado a MANTENIMIENTO
      if(datos.falla === 'SI') {
        actualizarEstadoUnidad(datos.unidad, 'MANTENIMIENTO');
        registrarTrazabilidad(datos.unidad, data[i][0], 'FALLA', datos.descFalla || 'Falla reportada', datos.pin);
      } else {
        actualizarEstadoUnidad(datos.unidad, 'DISPONIBLE');
      }
      
      registrarTrazabilidad(datos.unidad, data[i][0], 'ENTRADA', 'Entrada registrada', datos.pin);
      
      return { exito: true, mensaje: 'Entrada registrada exitosamente' };
    }
  }
  
  return { exito: false, error: 'No se encontró viaje abierto para esta unidad' };
}

/**
 * MODAL 4: REGISTRO DE DIESEL
 * Registra carga de combustible con foto de ticket
 */
function registrarDiesel(datos, fotoBase64) {
  const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  let s = ss.getSheetByName('DIESEL');
  
  if(!s) {
    s = ss.insertSheet('DIESEL');
    s.appendRow(['ID', 'FECHA', 'UNIDAD', 'CHOFER_PIN', 'LITROS', 'COSTO', 'KM', 'FOTO_TICKET']);
  }
  
  const id = 'DSL-' + new Date().getTime();
  const fecha = new Date();
  
  s.appendRow([
    id,
    fecha,
    datos.unidad,
    datos.pin,
    datos.litros,
    datos.costo,
    datos.km,
    fotoBase64 || ''
  ]);
  
  // Calcular rendimiento si hay datos previos
  const sV = ss.getSheetByName('VIAJES');
  if(sV) {
    const vData = sV.getDataRange().getValues();
    let kmAnterior = 0;
    
    for(let i = vData.length - 1; i > 0; i--) {
      if(vData[i][2] === datos.unidad && vData[i][11] === 'CERRADO' && vData[i][7]) {
        kmAnterior = vData[i][7];
        break;
      }
    }
    
    if(kmAnterior > 0) {
      const kmRecorridos = datos.km - kmAnterior;
      const rendimiento = (kmRecorridos / datos.litros).toFixed(2);
      registrarTrazabilidad(datos.unidad, id, 'DIESEL', `Carga: ${datos.litros}L - Rendimiento: ${rendimiento} km/L`, datos.pin);
    } else {
      registrarTrazabilidad(datos.unidad, id, 'DIESEL', `Carga: ${datos.litros}L - $${datos.costo}`, datos.pin);
    }
  }
  
  return { exito: true, mensaje: 'Carga de diesel registrada' };
}

          estado: row[7] || 'PENDIENTE',
          insumos: row[8] ? row[8].split(',').map(i => i.trim()) : [],
          refacciones: row[9] ? row[9].split(',').map(r => r.trim()) : [],
          observaciones: row[10] || ''
        };
      }
    }
    
    return null;
    
  } catch(e) {
    Logger.log('Error en obtenerDetalleOT: ' + e.message);
    return null;
  }
}

/**
 * Cierra una orden de trabajo
 */
function cerrarOrdenTrabajo(folio, diagnostico, observaciones) {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sh = ss.getSheetByName('ORDENES_TRABAJO');
    
    if(!sh) {
      return { exito: false, error: 'Hoja ORDENES_TRABAJO no encontrada' };
    }
    
    const data = sh.getDataRange().getValues();
    
    for(let i = 1; i < data.length; i++) {
      const row = data[i];
      if(row[0] == folio) {
        // Actualizar diagnóstico, observaciones y estado
        sh.getRange(i + 1, 7).setValue(diagnostico); // Diagnóstico
        sh.getRange(i + 1, 8).setValue('CERRADA'); // Estado
        sh.getRange(i + 1, 11).setValue(observaciones || ''); // Observaciones
        sh.getRange(i + 1, 12).setValue(new Date()); // Fecha de cierre
        
        // Registrar en trazabilidad
        registrarTrazabilidad('TALLER', 'CERRAR_OT', `OT ${folio} cerrada`);
        
        return { exito: true, mensaje: 'OT cerrada correctamente' };
      }
    }
    
    return { exito: false, error: 'OT no encontrada' };
    
  } catch(e) {
    Logger.log('Error en cerrarOrdenTrabajo: ' + e.message);
    return { exito: false, error: e.message };
  }
}

/**
 * Obtiene las unidades disponibles para asignación
 */
function obtenerUnidadesDisponibles() {
  try {
    const ss = SpreadsheetApp.openById(SPREADSHEET_ID);
    const sh = ss.getSheetByName('FLOTA');
    
    if(!sh) {
      return [];
    }
    
    const data = sh.getDataRange().getValues();
    const unidades = [];
    
    for(let i = 1; i < data.length; i++) {
      const row = data[i];
      if(row[7] === 'DISPONIBLE' || row[7] === 'ACTIVA') {
        unidades.push({
          id: row[0],
          modelo: row[1],
          placas: row[2],
          estado: row[7]
        });
      }
    }
    
    return unidades;
    
  } catch(e) {
    Logger.log('Error en obtenerUnidadesDisponibles: ' + e.message);
    return [];
  }
}

/**

    if(!sh) {
      return { exito: false, error: 'Hoja SALIDAS no encontrada' };

    const usuario = Session.getActiveUser().getEmail();
    
    sh.appendRow([
      datos.unidad,
      new Date(),
      usuario,
      datos.destino,
      datos.km,
      datos.obs || '',
      'SALIDA'
    ]);
    
    // Actualizar estado de la unidad
    const shFlota = ss.getSheetByName('FLOTA');
    if(shFlota) {
      const dataFlota = shFlota.getDataRange().getValues();
      for(let i = 1; i < dataFlota.length; i++) {
        if(dataFlota[i][0] === datos.unidad) {
          shFlota.getRange(i + 1, 8).setValue('EN_RUTA');
          break;
        }
      }
    }
    
    registrarTrazabilidad(usuario, 'SALIDA', `Salida de ${datos.unidad} hacia ${datos.destino}`);
    
    return { exito: true, mensaje: 'Salida registrada' };
    
  } catch(e) {
    Logger.log('Error en registrarSalida: ' + e.message);
    return { exito: false, error: e.message };
  }
}
